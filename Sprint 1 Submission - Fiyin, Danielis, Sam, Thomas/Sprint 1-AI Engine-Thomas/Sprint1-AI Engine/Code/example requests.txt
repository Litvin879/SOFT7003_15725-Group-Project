start the application: uvicorn app.main:app --reload

Open the interactive API documentation at:

http://127.0.0.1:8000/docs

US1 — Submit property details
curl -X POST http://127.0.0.1:8000/properties \
  -H "Content-Type: application/json" \
  -d '{
    "location": "Killarney, County Kerry",
    "bedrooms": 3,
    "property_type": "detached_house",
    "ber_rating": "B2"
  }'

US3 — Assess affordability
curl -X POST http://127.0.0.1:8000/affordability/assess \
  -H "Content-Type: application/json" \
  -d '{
    "application_id": "APP-001",
    "annual_income": 65000,
    "annual_debt_repayments": 12000,
    "deposit": 40000,
    "mortgage_term_years": 30
  }'

US5 — Run the valuation agent
curl -X POST http://127.0.0.1:8000/agents/valuation \
  -H "Content-Type: application/json" \
  -d '{
    "location": "Killarney, County Kerry",
    "bedrooms": 3,
    "property_type": "detached_house",
    "ber_rating": "B2"
  }'
